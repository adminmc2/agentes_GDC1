% --- Fuentes del sistema (xelatex) ---

% --- Geometría de la página ---
\usepackage[a4paper,
            top=2.5cm, bottom=2.5cm,
            left=2.5cm, right=2.5cm,
            footskip=1.2cm,
            headheight=14pt]{geometry}

% --- Colores: paleta NC1 ---

% --- Interlineado y párrafo ---
\linespread{1.20}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.5em plus 0.1em}

% --- Cabecera y pie (paquete core) ---
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\textcolor{muted}{\small\textsc{Diagnóstico NC1}}}
\fancyhead[R]{\textcolor{muted}{\small\nouppercase{\leftmark}}}

\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}
\renewcommand{\headrule}{\hbox to\headwidth{\color{border}\leaders\hrule height \headrulewidth\hfill}}
\renewcommand{\sectionmark}[1]{\markboth{\textcolor{accent}{#1}}{}}

% --- Estilos de section/subsection/etc (sin titlesec) ---
\makeatletter
\renewcommand{\section}{\@startsection{section}{1}{0pt}
  {-1.6em \@plus -0.3em \@minus -0.2em}
  {0.9em \@plus 0.2em}
  {\normalfont\Large\bfseries\color{accent}}}
\renewcommand{\subsection}{\@startsection{subsection}{2}{0pt}
  {-1.2em \@plus -0.3em \@minus -0.2em}
  {0.55em \@plus 0.2em}
  {\normalfont\large\bfseries\color{accent}}}
\renewcommand{\subsubsection}{\@startsection{subsubsection}{3}{0pt}
  {-0.9em \@plus -0.3em \@minus -0.2em}
  {0.4em \@plus 0.2em}
  {\normalfont\normalsize\bfseries\color{accent}}}
\renewcommand{\paragraph}{\@startsection{paragraph}{4}{0pt}
  {-0.7em \@plus -0.3em \@minus -0.2em}
  {-0.6em}
  {\normalfont\normalsize\bfseries\color{accent}}}
\makeatother

% --- Listas: ajusta separación con \parsep / \itemsep ---
\makeatletter
\def\@listi{\leftmargin\leftmargini
            \parsep 0pt
            \topsep 0.3em
            \itemsep 0.25em}
\let\@listI\@listi
\@listi
\makeatother

% --- Citas: cursivas, color suave (sin paquetes contribuidos) ---
\let\oldquote\quote

\renewenvironment{quote}{\oldquote\itshape\color{muted}\small}{\oldendquote}
